cmake_minimum_required(VERSION 3.10)
project(NDI_CPP_Test)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Find NDI SDK
set(NDI_SDK_PATH "C:/Program Files/NDI/NDI 6 SDK")
set(NDI_INCLUDE_DIR "${NDI_SDK_PATH}/Include")
set(NDI_LIB_DIR "${NDI_SDK_PATH}/Lib/x64")

# Check if NDI SDK exists
if(NOT EXISTS "${NDI_INCLUDE_DIR}/Processing.NDI.Lib.h")
    message(FATAL_ERROR "NDI SDK not found at ${NDI_SDK_PATH}")
endif()

# Include directories
include_directories(${NDI_INCLUDE_DIR})

# Add executable
add_executable(ndi_cpp_test ndi_cpp_test.cpp)

# Link libraries
target_link_libraries(ndi_cpp_test 
    "${NDI_LIB_DIR}/Processing.NDI.Lib.x64.lib"
)

# Copy NDI DLL to output directory
add_custom_command(TARGET ndi_cpp_test POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${CMAKE_CURRENT_SOURCE_DIR}/Processing.NDI.Lib.x64.dll"
    $<TARGET_FILE_DIR:ndi_cpp_test>
)

# Set output directory
set_target_properties(ndi_cpp_test PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin"
)

message(STATUS "NDI SDK found at: ${NDI_SDK_PATH}")
message(STATUS "Include directory: ${NDI_INCLUDE_DIR}")
message(STATUS "Library directory: ${NDI_LIB_DIR}")
